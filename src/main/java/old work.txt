    public static int Distance(Node<Integer>L1,int a)
    {
        Node<Integer> p=L1;
        int length=0;
        int Start_index=0;
        int End_index=0;

        boolean st=false;

        {
            return -1;
        }
        while (p!=null)
        {
            length++;
            if(p.getValue().equals(a) && !st)
            {
                Start_index=length;
                st=!st;
            }
            if(p.getValue().equals(a) && st)
            {
                End_index=length;
            }
            p=p.getNext();
                
        }
        Start_index--;
        End_index--;

        return Start_index+length-End_index-1;

    }
